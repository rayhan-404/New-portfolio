  // AUDIO FX ENGINE (Web Audio Buffer Decoder + Base64 MP3 Multi-Player)

  let soundEnabled = localStorage.getItem('portfolio_sound_fx') !== 'disabled';

  const decodedAudioBuffers = {};

  function getAudioContext() {
    if (!sharedAudioCtx) {
      const AudioCtx = window.AudioContext || window.webkitAudioContext;
      if (AudioCtx) {
        sharedAudioCtx = new AudioCtx();
      }
    }
    if (sharedAudioCtx && sharedAudioCtx.state === 'suspended') {

    }
    return sharedAudioCtx;

  function base64ToArrayBuffer(base64Uri) {
    const binaryString = window.atob(base64Uri.split(',')[1] || base64Uri);
    const len = binaryString.length;
    const bytes = new Uint8Array(len);
    for (let i = 0; i < len; i++) {
      bytes[i] = binaryString.charCodeAt(i);
    }
    return bytes.buffer;
  }

  // Pre-decode Base64 MP3 sounds into Web Audio buffers
  function initAudioEngine() {
    const ctx = getAudioContext();
    if (!window.SOUNDS_BASE64) return;

    Object.keys(window.SOUNDS_BASE64).forEach(key => {
      const dataUri = window.SOUNDS_BASE64[key];
      // Prepare HTML5 Audio fallback pool
      try {
        const a = new Audio(dataUri);
        a.preload = 'auto';
        htmlAudioPool[key] = a;
      } catch (e) {}

      // Decode into Web Audio buffer for infallible instant play
      if (ctx && ctx.decodeAudioData) {
        try {
          const ab = base64ToArrayBuffer(dataUri);
          ctx.decodeAudioData(ab, (buffer) => {
            decodedAudioBuffers[key] = buffer;
          }, () => {});
        } catch (e) {}
      }
    });
  }

  // Auto initialize on first interaction
  const unlockAudioEngine = () => {
    const ctx = getAudioContext();
    if (ctx && ctx.state === 'suspended') {
      ctx.resume().catch(() => {});
    }

    ['click', 'touchstart', 'pointerdown', 'keydown'].forEach(evt => {
      window.removeEventListener(evt, unlockAudioEngine);
    });
  };
  ['click', 'touchstart', 'pointerdown', 'keydown'].forEach(evt => {
    window.addEventListener(evt, unlockAudioEngine, { passive: true, once: true });
  });

  // Attempt initial load

  function playSound(key, volume = 0.5) {
    if (!soundEnabled) return;
    let played = false;

    // 1. Primary: Web Audio AudioBufferSource (fastest, no autoplay clipping)
    try {
      const ctx = getAudioContext();
      if (ctx && decodedAudioBuffers[key]) {
        if (ctx.state === 'suspended') {
          ctx.resume().catch(() => {});
        }
        const source = ctx.createBufferSource();
        source.buffer = decodedAudioBuffers[key];
        const gainNode = ctx.createGain();
        gainNode.gain.value = volume;
        source.connect(gainNode);
        gainNode.connect(ctx.destination);
        source.start(0);
        played = true;
      }
    } catch (e) {}

    // 2. Fallback: Base64 HTML5 Audio Element
    if (!played) {
      try {
        const poolItem = htmlAudioPool[key];
        const dataUri = window.SOUNDS_BASE64 ? window.SOUNDS_BASE64[key] : ('/sounds/' + key + '.mp3');
        const audio = poolItem ? poolItem.cloneNode() : new Audio(dataUri);
        audio.volume = volume;
        const p = audio.play();
        if (p !== undefined) {
          p.catch(() => {});
        }
      } catch (e) {}
    }
  }

  // High-Performance Scroll Spy (Two-way scroll detection)
  function handleScrollSpy() {
    if (isProgrammaticScrolling) return;

    const containerHeight = contentContainer.clientHeight;
    let currentSectionIndex = 0;

    sections.forEach((sec, idx) => {
      const rect = sec.getBoundingClientRect();
      const containerRect = contentContainer.getBoundingClientRect();
      const relativeTop = rect.top - containerRect.top;
      
      if (relativeTop <= 180 && relativeTop >= -sec.offsetHeight + 180) {
        currentSectionIndex = idx;
      }
    });

    // Special case for bottom of scroll (Contact section)
    if (contentContainer.scrollHeight - contentContainer.scrollTop - containerHeight < 70) {
      currentSectionIndex = sections.length - 1;
    }

    if (currentSectionIndex !== activeIndex) {
      updateActiveNav(currentSectionIndex, true);
    }
  }

  function initSandboxCanvas() {
    const cvs = document.getElementById('interactive-draw-canvas');
    if (!cvs) return;
    const ctx = cvs.getContext('2d');
    let isDrawing = false;
    let lastX = 0;
    let lastY = 0;

    // Draw initial welcome signature
    ctx.strokeStyle = '#0d4c3c';
    ctx.lineWidth = 3;
    ctx.lineCap = 'round';
    ctx.lineJoin = 'round';
    ctx.beginPath();
    ctx.moveTo(30, 60);
    ctx.bezierCurveTo(80, 20, 120, 90, 180, 50);
    ctx.stroke();

    function getCoords(e) {
      const rect = cvs.getBoundingClientRect();
      const clientX = e.touches ? e.touches[0].clientX : e.clientX;
      const clientY = e.touches ? e.touches[0].clientY : e.clientY;
      return {
        x: (clientX - rect.left) * (cvs.width / rect.width),
        y: (clientY - rect.top) * (cvs.height / rect.height)
      };
    }

    function startDraw(e) {
      isDrawing = true;
      const c = getCoords(e);
      lastX = c.x;
      lastY = c.y;
      e.preventDefault();
    }

    function draw(e) {
      if (!isDrawing) return;
      const c = getCoords(e);
      ctx.strokeStyle = canvasPenColor;
      ctx.lineWidth = 3;
      ctx.beginPath();
      ctx.moveTo(lastX, lastY);
      ctx.lineTo(c.x, c.y);
      ctx.stroke();
      lastX = c.x;
      lastY = c.y;
      e.preventDefault();
    }

    function stopDraw() { isDrawing = false; }

    cvs.addEventListener('mousedown', startDraw);
    cvs.addEventListener('mousemove', draw);
    window.addEventListener('mouseup', stopDraw);

    cvs.addEventListener('touchstart', startDraw, { passive: false });
    cvs.addEventListener('touchmove', draw, { passive: false });
    window.addEventListener('touchend', stopDraw);
  }

  // Staggered Entrance Animation via IntersectionObserver
  const revealElements = document.querySelectorAll('.reveal-item');
  
  if ('IntersectionObserver' in window) {
    const revealObserver = new IntersectionObserver((entries, observer) => {
      entries.forEach(entry => {
        if (entry.isIntersecting) {
          entry.target.classList.add('in-view');
          observer.unobserve(entry.target);
        }
      });
    }, {
      root: contentContainer,
      rootMargin: '0px 0px -20px 0px',
      threshold: 0.05
    });

    revealElements.forEach(el => revealObserver.observe(el));

    // Instant initial check for visible elements upon load to avoid animation delays on initial screen
    requestAnimationFrame(() => {
      revealElements.forEach(el => {
        const rect = el.getBoundingClientRect();
        const containerRect = contentContainer.getBoundingClientRect();
        if (rect.top < containerRect.bottom && rect.bottom > containerRect.top) {
          el.classList.add('in-view');
          revealObserver.unobserve(el);
        }
      });
    });
  } else {
    // Graceful fallback for non-supporting browsers
    revealElements.forEach(el => el.classList.add('in-view'));

  // =========================================================
  // FRAMER MOTION-INSPIRED HERO SVG 3D PARALLAX & SPRING PHYSICS
  // =========================================================
  const heroCircle = document.getElementById('hero-interactive-circle');
  const heroVisualCol = document.querySelector('.section-visual-col');
  const layerGlow = document.getElementById('hero-layer-glow');
  const layerBg = document.getElementById('hero-layer-bg');
  const layerBody = document.getElementById('hero-layer-body');
  const layerHead = document.getElementById('hero-layer-head');
  const layerFace = document.getElementById('hero-layer-face');
  const layerBadge = document.getElementById('hero-layer-badge');

  if (heroCircle && heroVisualCol) {
    // Spring physics state
    let targetRotateX = 0;
    let targetRotateY = 0;
    let targetTranslateX = 0;
    let targetTranslateY = 0;
    let targetScale = 1;

    let currentRotateX = 0;
    let currentRotateY = 0;
    let currentTranslateX = 0;
    let currentTranslateY = 0;
    let currentScale = 1;

    let isHovering = false;
    let animationLoopId = null;

    const springDamping = 0.09; // Framer motion smooth spring response

    function updateHeroMotion() {
      // Spring interpolation (lerp)
      currentRotateX += (targetRotateX - currentRotateX) * springDamping;
      currentRotateY += (targetRotateY - currentRotateY) * springDamping;
      currentTranslateX += (targetTranslateX - currentTranslateX) * springDamping;
      currentTranslateY += (targetTranslateY - currentTranslateY) * springDamping;
      currentScale += (targetScale - currentScale) * springDamping;

      // Apply 3D perspective transform to hero circle container
      heroCircle.style.transform = `perspective(700px) rotateX(${currentRotateX.toFixed(2)}deg) rotateY(${currentRotateY.toFixed(2)}deg) translate3d(${currentTranslateX.toFixed(2)}px, ${currentTranslateY.toFixed(2)}px, 0) scale(${currentScale.toFixed(3)})`;

      // Multi-layer depth parallax inside SVG
      const normX = currentRotateY / 14; // -1 to 1
      const normY = -currentRotateX / 12; // -1 to 1

      if (layerGlow) layerGlow.style.transform = `translate(${-normX * 4}px, ${-normY * 4}px)`;
      if (layerBg) layerBg.style.transform = `translate(${-normX * 2}px, ${-normY * 2}px)`;
      if (layerBody) layerBody.style.transform = `translate(${normX * 4}px, ${normY * 4}px)`;
      if (layerHead) layerHead.style.transform = `translate(${normX * 8}px, ${normY * 8}px)`;
      if (layerFace) layerFace.style.transform = `translate(${normX * 13}px, ${normY * 11}px)`;
      if (layerBadge) layerBadge.style.transform = `translate(${normX * 18}px, ${normY * 15}px)`;

      // Continue loop if moving or hovering
      const isResting = Math.abs(targetRotateX - currentRotateX) < 0.01 && 
                        Math.abs(targetRotateY - currentRotateY) < 0.01 && 
                        Math.abs(targetTranslateX - currentTranslateX) < 0.01 && 
                        Math.abs(targetTranslateY - currentTranslateY) < 0.01 &&
                        Math.abs(targetScale - currentScale) < 0.001;

      if (!isResting || isHovering) {
        animationLoopId = requestAnimationFrame(updateHeroMotion);
      } else {
        animationLoopId = null;
      }
    }

    function startLoop() {
      if (!animationLoopId) {
        animationLoopId = requestAnimationFrame(updateHeroMotion);
      }
    }

    const onPointerMove = (e) => {
      isHovering = true;
      const rect = heroVisualCol.getBoundingClientRect();
      const centerX = rect.left + rect.width / 2;
      const centerY = rect.top + rect.height / 2;

      const deltaX = (e.clientX - centerX) / (rect.width / 2);
      const deltaY = (e.clientY - centerY) / (rect.height / 2);

      const clampedX = Math.max(-1, Math.min(1, deltaX));
      const clampedY = Math.max(-1, Math.min(1, deltaY));

      // Calculate Framer Motion 3D tilt & shift
      targetRotateY = clampedX * 14; // max 14deg tilt left/right
      targetRotateX = -clampedY * 12; // max 12deg tilt up/down
      targetTranslateX = clampedX * 7;
      targetTranslateY = clampedY * 7;
      targetScale = 1.06;

      startLoop();
    };

    const onPointerLeave = () => {
      isHovering = false;
      targetRotateX = 0;
      targetRotateY = 0;
      targetTranslateX = 0;
      targetTranslateY = 0;
      targetScale = 1;
      startLoop();
    };

    heroVisualCol.addEventListener('pointermove', onPointerMove, { passive: true });
    heroVisualCol.addEventListener('pointerleave', onPointerLeave, { passive: true });

    // Interactive tactile pop on clicking the hero illustration
    heroCircle.addEventListener('click', () => {
      targetScale = 0.94;
      soundFx.pop();
      showToast("Hey there! Welcome to Rayhan's portfolio 👋", "✨");
      setTimeout(() => {
        targetScale = isHovering ? 1.06 : 1;
      }, 120);
    });
  }
